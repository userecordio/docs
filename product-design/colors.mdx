export const ColorChip = ({ hex, height, label }) => {
  const [copied, setCopied] = useState(false);
  const copy = () => {
    try { navigator.clipboard.writeText(hex); } catch (e) {}
    setCopied(true);
    setTimeout(() => setCopied(false), 1200);
  };
  return (
    <button type="button" onClick={copy} aria-label={"Copy " + hex} title={hex}
      style={{ display: "block", width: "100%", height: height || 56, background: hex, border: "1px solid rgba(127,127,127,.25)", borderRadius: 8, cursor: "pointer", padding: 0, position: "relative" }}>
      {copied && <span style={{ position: "absolute", inset: 0, display: "flex", alignItems: "center", justifyContent: "center", fontSize: 11, fontWeight: 600, color: "#fff", background: "rgba(0,0,0,.55)", borderRadius: 7 }}>Copied</span>}
    </button>
  );
};
