Most components use **6px** (`rounded-md`). Use the class, not the pixel value.

```tsx
<div className="rounded-md border">…</div>
```

## Border width

| Class | Value |
| --- | --- |
| `border-0` | 0 |
| `border` | 1px |
| `border-2` | 2px |
| `border-4` | 4px |
| `border-8` | 8px |

<Accordion title="All radius values as a table">

</Accordion>

export const RadiusScale = () => {
  const R = [["rounded-none", 0], ["rounded-sm", 2], ["rounded", 4], ["rounded-md", 6, "Default"], ["rounded-lg", 8], ["rounded-xl", 12], ["rounded-2xl", 16], ["rounded-3xl", 24], ["rounded-4xl", 32], ["rounded-full", 9999]];
  return (
    <div className="not-prose" style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(140px, 1fr))", gap: 16 }}>
      {R.map(([cls, px, tag]) => (
        <div key={cls} style={{ border: "1px solid rgba(127,127,127,.25)", borderRadius: 12, padding: 16, display: "grid", justifyItems: "center", gap: 12, minWidth: 0 }}>
          <div style={{ width: 72, height: 72, background: "rgba(255,87,35,.12)", border: "2px solid #ff5723", borderRadius: px }} />
          <div style={{ textAlign: "center", display: "grid", gap: 2 }}>
            <div style={{ fontSize: 13, fontWeight: 600, fontFamily: "ui-monospace, Menlo, monospace" }}>{cls}</div>
            <div style={{ fontSize: 12, opacity: .65, fontFamily: "ui-monospace, Menlo, monospace" }}>{px === 9999 ? "9999px" : px + "px"}{tag ? " · " + tag : ""}</div>
          </div>
        </div>
      ))}
    </div>
  );
};

export const ComponentRadius = () => {
  const C = [["Button", "rounded-md", 6], ["Input", "rounded-md", 6], ["Badge", "rounded-md", 6], ["Toggle", "rounded-md", 6], ["Dropdown Menu", "rounded-md", 6], ["Pagination item", "rounded-md", 6], ["Tabs list", "rounded-lg", 8], ["Checkbox", "rounded-sm", 2], ["Table cell", "rounded-none", 0]];
  const row = { display: "grid", gridTemplateColumns: "minmax(0,1fr) 120px 56px 40px", gap: 16, alignItems: "center", padding: "10px 16px" };
  const mono = { fontSize: 13, fontFamily: "ui-monospace, Menlo, monospace" };
  return (
    <div className="not-prose" style={{ border: "1px solid rgba(127,127,127,.25)", borderRadius: 12, overflow: "hidden" }}>
      <div style={{ ...row, background: "rgba(127,127,127,.06)", fontSize: 11, fontWeight: 600, textTransform: "uppercase", letterSpacing: ".04em", opacity: .7 }}>
        <span>Component</span><span>Class</span><span>Value</span><span />
      </div>
      {C.map(([name, cls, px]) => (
        <div key={name} style={{ ...row, borderTop: "1px solid rgba(127,127,127,.15)" }}>
          <span style={{ fontSize: 14, fontWeight: 500 }}>{name}</span>
          <span style={mono}>{cls}</span>
          <span style={{ ...mono, opacity: .7 }}>{px}px</span>
          <div style={{ width: 28, height: 20, border: "2px solid #ff5723", borderRadius: px }} />
        </div>
      ))}
    </div>
  );
};
