One step is **4px**. The number in the class is the number of steps: `p-4` is 16px, `gap-2` is 8px.

```tsx
<div className="flex flex-col gap-4 p-6">…</div>
```

export const SpacingScale = () => {
  const S = [["0", 0], ["px", 1], ["0.5", 2], ["1", 4], ["1.5", 6], ["2", 8], ["2.5", 10], ["3", 12], ["3.5", 14], ["4", 16], ["5", 20], ["6", 24], ["7", 28], ["8", 32], ["9", 36], ["10", 40], ["11", 44], ["12", 48], ["14", 56], ["16", 64], ["20", 80], ["24", 96], ["28", 112], ["32", 128], ["36", 144], ["40", 160], ["44", 176], ["48", 192], ["52", 208], ["56", 224], ["60", 240], ["64", 256], ["72", 288], ["80", 320], ["96", 384]];
  const row = { display: "grid", gridTemplateColumns: "64px 64px minmax(0,1fr)", gap: 16, alignItems: "center", padding: "8px 16px" };
  const mono = { fontSize: 13, fontFamily: "ui-monospace, Menlo, monospace" };
  return (
    <div className="not-prose" style={{ border: "1px solid rgba(127,127,127,.25)", borderRadius: 12, overflow: "hidden" }}>
      <div style={{ ...row, background: "rgba(127,127,127,.06)", fontSize: 11, fontWeight: 600, textTransform: "uppercase", letterSpacing: ".04em", opacity: .7 }}>
        <span>Step</span><span>Value</span><span>Size</span>
      </div>
      {S.map(([k, px]) => (
        <div key={k} style={{ ...row, borderTop: "1px solid rgba(127,127,127,.15)" }}>
          <span style={{ ...mono, fontWeight: 600 }}>{k}</span>
          <span style={{ ...mono, opacity: .7 }}>{px}px</span>
          <div style={{ height: 12, width: Math.max(px, 1), maxWidth: "100%", background: "#ff5723", borderRadius: 2, opacity: px ? 1 : .3 }} />
        </div>
      ))}
    </div>
  );
};

export const SizeTable = ({ kind }) => {
  const MAXW = [["max-w-3xs", 256], ["max-w-2xs", 288], ["max-w-xs", 320], ["max-w-sm", 384], ["max-w-md", 448], ["max-w-lg", 512], ["max-w-xl", 576], ["max-w-2xl", 672], ["max-w-3xl", 768], ["max-w-4xl", 896], ["max-w-5xl", 1024], ["max-w-6xl", 1152], ["max-w-7xl", 1280]];
  const BP = [["sm", 640], ["md", 768], ["lg", 1024], ["xl", 1280], ["2xl", 1536]];
  const rows = kind === "bp" ? BP : MAXW;
  const max = rows[rows.length - 1][1];
  const row = { display: "grid", gridTemplateColumns: "110px 72px minmax(0,1fr)", gap: 16, alignItems: "center", padding: "8px 16px" };
  const mono = { fontSize: 13, fontFamily: "ui-monospace, Menlo, monospace" };
  return (
    <div className="not-prose" style={{ border: "1px solid rgba(127,127,127,.25)", borderRadius: 12, overflow: "hidden" }}>
      {rows.map(([k, px], i) => (
        <div key={k} style={{ ...row, borderTop: i ? "1px solid rgba(127,127,127,.15)" : "none" }}>
          <span style={{ ...mono, fontWeight: 600 }}>{k}</span>
          <span style={{ ...mono, opacity: .7 }}>{px}px</span>
          <div style={{ height: 8, width: (px / max * 100) + "%", background: "rgba(127,127,127,.35)", borderRadius: 4 }} />
        </div>
      ))}
    </div>
  );
};
